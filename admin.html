Aseguramos la columna por si ya tenías la tabla medio creada ALTER TABLE public.categories ADD COLUMN IF NOT EXISTS is_visible boolean DEFAULT true; -- 3. Insertamos categorías por defecto (si no existen) INSERT INTO public.categories (name) VALUES ('Camisetas'), ('Sudaderas'), ('Pantalones'), ('Gorras'), ('Accesorios') ON CONFLICT (name) DO NOTHING; -- 4. Enlazamos los productos a las categorías ALTER TABLE public.products ADD COLUMN IF NOT EXISTS category_id integer REFERENCES public.categories(id) ON DELETE SET NULL; -- 5.
